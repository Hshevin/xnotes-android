function(pdfium_library name kind)
    if(name MATCHES "^third_party_cpu_features_")
        return()
    endif()
    cmake_parse_arguments(PARSE_ARGV 2 GROUP "" "" "SOURCES;DEFINES;INCLUDES;FLAGS;CFLAGS;CXXFLAGS")
    list(FILTER GROUP_SOURCES EXCLUDE REGEX "^core/fxge/android/")
    list(TRANSFORM GROUP_SOURCES PREPEND "${PDFIUM}/")
    list(TRANSFORM GROUP_INCLUDES PREPEND "${PDFIUM}/")
    list(TRANSFORM GROUP_DEFINES REPLACE "^ARMV8_OS_ANDROID$" "ARMV8_OS_MACOS")
    if(name STREQUAL core_fxge)
        list(APPEND GROUP_SOURCES "${APPLE_SOURCES}/core/fxge/apple/capple_platform.cpp"
            "${APPLE_SOURCES}/core/fxge/apple/cquartz_2d.cpp" "${APPLE_SOURCES}/core/fxge/apple/fx_apple_impl.cpp")
    endif()
    add_library(${name} OBJECT ${GROUP_SOURCES})
    target_include_directories(${name} PRIVATE "${APPLE_SOURCES}" ${GROUP_INCLUDES})
    target_compile_definitions(${name} PRIVATE NDEBUG NVALGRIND DYNAMIC_ANNOTATIONS_ENABLED=0 ${GROUP_DEFINES})
    target_compile_options(${name} PRIVATE -w -fno-strict-aliasing -ffp-contract=off -ftrivial-auto-var-init=zero
        ${GROUP_FLAGS} "$<$<COMPILE_LANGUAGE:C>:${GROUP_CFLAGS}>"
        "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;${GROUP_CXXFLAGS}>")
    target_sources(xnotespdf PRIVATE $<TARGET_OBJECTS:${name}>)
endfunction()
